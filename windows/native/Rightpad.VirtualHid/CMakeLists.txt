cmake_minimum_required(VERSION 3.24)
project(RightpadVirtualHid LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "rightpad Virtual HID POC requires Windows x64")
endif()
include(FetchContent)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBVIRTUALHID_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LIBVIRTUALHID_BUILD_WINDOWS_DRIVER OFF CACHE BOOL "" FORCE)
set(LIBVIRTUALHID_BUILD_WINDOWS_BROKER OFF CACHE BOOL "" FORCE)
set(LIBVIRTUALHID_INSTALL OFF CACHE BOOL "" FORCE)
set(LIBVIRTUALHID_ENABLE_PACKAGING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libvirtualhid
    GIT_REPOSITORY https://github.com/LizardByte/libvirtualhid.git
    GIT_TAG 6fdb8bd4de3b68d96c30e5303ac2ebb333c09746
    GIT_SUBMODULES third-party/lizardbyte-common)
FetchContent_MakeAvailable(libvirtualhid)
add_library(Rightpad.VirtualHid SHARED bridge.cpp)
target_compile_definitions(Rightpad.VirtualHid PRIVATE RIGHTPAD_VHID_EXPORTS)
target_link_libraries(Rightpad.VirtualHid PRIVATE libvirtualhid::libvirtualhid)
set_target_properties(Rightpad.VirtualHid PROPERTIES PREFIX "")
if(MINGW)
    # Ship one bridge DLL; avoid a runtime dependency on an installed MinGW tree.
    target_link_options(Rightpad.VirtualHid PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(bridge_tests bridge_tests.cpp)
target_link_libraries(bridge_tests PRIVATE Rightpad.VirtualHid libvirtualhid::libvirtualhid)
if(MINGW)
    target_link_options(bridge_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME bridge_fake_api COMMAND bridge_tests)
